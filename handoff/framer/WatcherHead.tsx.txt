// WatcherHead — Framer code component for the Always Hungry Records homepage.
// The pinned cat head. Ink on light sections; flips to paper, line for line,
// wherever a dark section passes behind it.
//
// Setup in Framer:
// 1. Assets > Code > New component, paste this file.
// 2. Drop it on the page. Position: Fixed. Pin left 0, bottom 0.
//    Size: 516 x 620 (Desktop), 158 x 190 (Phone). Keep the 199:239 ratio.
// 3. Give every dark section frame an ID (right panel > Scroll Section / ID),
//    e.g. "what-we-do", and list those IDs in the Dark sections control.
// 4. Keep it above the content (z-index) and turn off pointer events if it
//    sits over links.

import { addPropertyControls, ControlType } from "framer"
import { useEffect, useRef } from "react"

function Mark({ color }: { color: string }) {
    return (
        <svg viewBox="14 11 199 239" preserveAspectRatio="xMinYMax meet" style={{ position: "absolute", inset: 0, width: "100%", height: "100%", display: "block" }}>
            <g fill={color}>
            <path fillRule="evenodd" d="M 14 34.1 L 41 11 L 63 39 L 91 64 L 117 42 L 144 29 L 169 49 L 192 79 L 206 117 L 213 153 L 205 193 L 186 229 L 170.6 250 L 14 250 Z M 87.13 168.56 C 108.58 146.45 139.45 147.5 155.39 167.58 C 135.65 190.72 104.26 190.53 87.13 168.56 Z M 22.95 148.08 C 46.99 133.93 73.86 141 83.05 162.92 C 61.13 177.77 34.26 172.12 22.95 148.08 Z" />
            <path d="M 128.64 153.84 C 130.18 162.93 119.88 180.08 111.13 182.99 C 109.59 173.89 119.89 156.75 128.64 153.84 Z" />
            <path d="M 63.25 144.54 C 62.55 153.73 49.11 167.17 39.92 167.87 C 40.63 158.68 54.06 145.25 63.25 144.54 Z" />
            </g>
        </svg>
    )
}

export default function WatcherHead(props) {
    const { darkSections, ink, paper, flip, style } = props
    const root = useRef<HTMLDivElement>(null)
    const paperLayer = useRef<HTMLDivElement>(null)

    useEffect(() => {
        if (!flip) return
        let raf = 0
        const update = () => {
            raf = 0
            const r = root.current?.getBoundingClientRect()
            const layer = paperLayer.current
            if (!r || !layer) return
            // Union of the overlap with every dark section, top-most band only.
            let best: string = "inset(100% 0 0 0)"
            for (const id of darkSections || []) {
                const el = document.getElementById(id)
                if (!el) continue
                const d = el.getBoundingClientRect()
                const t = Math.max(0, d.top - r.top)
                const b = Math.max(0, r.bottom - d.bottom)
                if (t < r.height && b < r.height) { best = `inset(${t}px 0 ${b}px 0)`; break }
            }
            layer.style.clipPath = best
        }
        const onScroll = () => { if (!raf) raf = requestAnimationFrame(update) }
        update()
        window.addEventListener("scroll", onScroll, { passive: true })
        window.addEventListener("resize", onScroll)
        return () => {
            window.removeEventListener("scroll", onScroll)
            window.removeEventListener("resize", onScroll)
            if (raf) cancelAnimationFrame(raf)
        }
    }, [flip, (darkSections || []).join("|")])

    return (
        <div ref={root} style={{ ...style, position: "relative", pointerEvents: "none" }}>
            <Mark color={ink} />
            <div ref={paperLayer} style={{ position: "absolute", inset: 0, clipPath: "inset(100% 0 0 0)" }}>
                <Mark color={paper} />
            </div>
        </div>
    )
}

WatcherHead.defaultProps = {
    darkSections: ["what-we-do"],
    ink: "#141310",
    paper: "#F7F5F0",
    flip: true,
    width: 516,
    height: 620,
}

addPropertyControls(WatcherHead, {
    flip: { type: ControlType.Boolean, title: "Flip on dark", defaultValue: true },
    darkSections: { type: ControlType.Array, title: "Dark sections", control: { type: ControlType.String }, defaultValue: ["what-we-do"] },
    ink: { type: ControlType.Color, title: "Ink", defaultValue: "#141310" },
    paper: { type: ControlType.Color, title: "Paper", defaultValue: "#F7F5F0" },
})
